cmake_minimum_required(VERSION 3.18)
project(fitfight_pose LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CAN_COMPILE_SME OFF CACHE INTERNAL "Unity NDK does not support SME streaming attributes" FORCE)
set(UNIT_TEST OFF CACHE BOOL "" FORCE)
set(LIBYUV_DISABLE_JPEG ON CACHE BOOL "" FORCE)
if(NOT ANDROID)
    # Universal host validation: Apple's clang does not accept upstream -march flags.
    # Production Android keeps runtime-dispatched NEON; host uses portable kernels on arm64.
    set(FF_SAVED_PROCESSOR "${CMAKE_SYSTEM_PROCESSOR}")
    set(CMAKE_SYSTEM_PROCESSOR x86_64)
    add_compile_definitions(LIBYUV_DISABLE_NEON LIBYUV_DISABLE_SVE LIBYUV_DISABLE_SME)
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../sdk/libyuv" "${CMAKE_CURRENT_BINARY_DIR}/libyuv" EXCLUDE_FROM_ALL)
if(NOT ANDROID)
    set(CMAKE_SYSTEM_PROCESSOR "${FF_SAVED_PROCESSOR}")
endif()
add_library(fitfight_pose SHARED fitfight_pose.cpp fitfight_camera.cpp)
target_link_libraries(fitfight_pose PRIVATE yuv)
target_include_directories(fitfight_pose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/libyuv/include")
if(ANDROID)
    find_package(ncnn CONFIG REQUIRED)
    target_link_libraries(fitfight_pose PRIVATE ncnn log)
    target_link_options(fitfight_pose PRIVATE -Wl,-z,max-page-size=16384)
else()
    target_include_directories(fitfight_pose PRIVATE "${NCNN_SDK}/ncnn.framework/Versions/A/Headers/ncnn")
    target_link_libraries(fitfight_pose PRIVATE "${NCNN_SDK}/ncnn.framework/ncnn" "${NCNN_SDK}/openmp.framework/openmp")
endif()
