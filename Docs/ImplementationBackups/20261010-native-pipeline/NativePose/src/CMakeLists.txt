cmake_minimum_required(VERSION 3.18)
project(fitfight_pose LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(fitfight_pose SHARED fitfight_pose.cpp)
if(ANDROID)
    find_package(ncnn CONFIG REQUIRED)
    target_link_libraries(fitfight_pose PRIVATE ncnn log)
    target_link_options(fitfight_pose PRIVATE -Wl,-z,max-page-size=16384)
else()
    target_include_directories(fitfight_pose PRIVATE "${NCNN_SDK}/ncnn.framework/Versions/A/Headers/ncnn")
    target_link_libraries(fitfight_pose PRIVATE "${NCNN_SDK}/ncnn.framework/ncnn" "${NCNN_SDK}/openmp.framework/openmp")
endif()
